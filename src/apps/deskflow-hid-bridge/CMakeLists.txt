set(_sdk_archive "${CMAKE_SOURCE_DIR}/third_party/native-hid/karabiner-client-sdk-8.7.0.tar.gz")
file(SHA256 "${_sdk_archive}" _sdk_sha256)
if(NOT _sdk_sha256 STREQUAL "50f857349496dc7cf8dd61684aaa3dfa7bbc84735603f8db014a19e166a6f55b")
  message(FATAL_ERROR "Native HID client SDK checksum mismatch")
endif()
set(_sdk "${CMAKE_CURRENT_BINARY_DIR}/sdk")
set(_adapter "${CMAKE_SOURCE_DIR}/third_party/native-hid/report-completion.hpp.inc")
file(SHA256 "${_adapter}" _adapter_sha256)
if(NOT EXISTS "${_sdk}/verified-${_sdk_sha256}-${_adapter_sha256}")
  file(MAKE_DIRECTORY "${_sdk}")
  file(ARCHIVE_EXTRACT INPUT "${_sdk_archive}" DESTINATION "${_sdk}")
  set(_client_header "${_sdk}/include/pqrs/karabiner/driverkit/virtual_hid_device_service/client.hpp")
  file(READ "${_client_header}" _client)
  file(READ "${_adapter}" _completion_adapter)
  set(_anchor "  void async_virtual_hid_keyboard_terminate() {")
  string(FIND "${_client}" "${_anchor}" _anchor_position)
  if(_anchor_position LESS 0)
    message(FATAL_ERROR "The native HID client adapter does not match the pinned SDK")
  endif()
  string(REPLACE "${_anchor}" "${_completion_adapter}\n${_anchor}" _client "${_client}")
  file(WRITE "${_client_header}" "${_client}")
  file(WRITE "${_sdk}/verified-${_sdk_sha256}-${_adapter_sha256}" "v8.7.0 client protocol 7 with acknowledged reports\n")
endif()

add_executable(deskflow-hid-bridge main.cpp)
set_target_properties(deskflow-hid-bridge PROPERTIES CXX_STANDARD 23 CXX_STANDARD_REQUIRED ON)
target_include_directories(deskflow-hid-bridge PRIVATE "${CMAKE_SOURCE_DIR}/src/lib")
target_include_directories(deskflow-hid-bridge SYSTEM PRIVATE "${_sdk}/include" "${_sdk}/vendor/vendor/include")
target_compile_options(deskflow-hid-bridge PRIVATE -Wno-deprecated-declarations)
target_link_libraries(deskflow-hid-bridge PRIVATE
  "-framework IOKit" "-framework CoreFoundation" "-framework SystemConfiguration" "-framework Security")
if(BUILD_OSX_BUNDLE)
  install(TARGETS deskflow-hid-bridge RUNTIME DESTINATION "Deskflow.app/Contents/Resources/native-keyboard")
endif()
